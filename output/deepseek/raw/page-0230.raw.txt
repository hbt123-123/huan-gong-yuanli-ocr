equation[[242, 70, 690, 108]]
\[u_{\mathrm{i}} = \frac{d_{c}^{2}\rho_{\mathrm{p}}u_{\mathrm{i}}^{2}}{18\mu r_{\mathrm{m}}} = \frac{(5.2\times 10^{-6})^{2}\times 4500\times 13.9^{2}}{18\times 3\times 10^{-5}\times 0.15} \mathrm{m / s} = 0.29 \mathrm{m / s}\]
